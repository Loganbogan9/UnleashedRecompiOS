set(UNLEASHED_RECOMP_IOS OFF)
if (CMAKE_SYSTEM_NAME STREQUAL "iOS" OR IOS)
	set(UNLEASHED_RECOMP_IOS ON)
endif()

if (NOT UNLEASHED_RECOMP_IOS)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/file_to_c)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/bc_diff)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/fshasher)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/x_decompress)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/XenonRecomp)
	if(APPLE)
		include("${CMAKE_SOURCE_DIR}/cmake/PrepareDependency.cmake")
		set(UNLEASHED_RECOMP_SHADER_TARGET_SDK "macosx" CACHE STRING "Apple SDK used by the native host to prepare game shaders")
		set_property(CACHE UNLEASHED_RECOMP_SHADER_TARGET_SDK PROPERTY STRINGS macosx iphoneos)
		if(UNLEASHED_RECOMP_SHADER_TARGET_SDK STREQUAL "iphoneos")
			set(UNLEASHED_RECOMP_SHADER_TARGET "air64-apple-ios16.0")
		elseif(UNLEASHED_RECOMP_SHADER_TARGET_SDK STREQUAL "macosx")
			set(UNLEASHED_RECOMP_SHADER_TARGET "air64-apple-macos13.0")
		else()
			message(FATAL_ERROR "UNLEASHED_RECOMP_SHADER_TARGET_SDK must be macosx or iphoneos")
		endif()
		unleashed_prepare_dependency(XENOS_SOURCE xenos "${UNLEASHED_RECOMP_TOOLS_ROOT}/XenosRecomp"
			PATCHES "${CMAKE_SOURCE_DIR}/patches/dependencies/xenos-air-target.patch" COPY_DIRS XenosRecomp)
		add_subdirectory("${XENOS_SOURCE}" "${CMAKE_BINARY_DIR}/tools/XenosRecomp")
		target_compile_definitions(XenosRecomp PRIVATE
			XENOS_RECOMP_AIR_SDK="${UNLEASHED_RECOMP_SHADER_TARGET_SDK}"
			XENOS_RECOMP_AIR_TARGET="${UNLEASHED_RECOMP_SHADER_TARGET}")
	else()
		add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/XenosRecomp)
	endif()
else()
	set(THIRDPARTY_ROOT ${UNLEASHED_RECOMP_TOOLS_ROOT}/XenonRecomp/thirdparty)
	add_subdirectory(${THIRDPARTY_ROOT}/disasm)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/XenonRecomp/XenonUtils)
endif()
